cmake_minimum_required(VERSION 3.16)
project(forge_storage VERSION 0.1.1 LANGUAGES C)
find_package(PkgConfig REQUIRED)
pkg_check_modules(STORAGE REQUIRED IMPORTED_TARGET libmicrohttpd json-c libcurl openssl)
add_library(forge_storage STATIC src/client.c)
target_compile_features(forge_storage PUBLIC c_std_11)
target_link_libraries(forge_storage PUBLIC PkgConfig::STORAGE)
target_compile_options(forge_storage PRIVATE -Wall -Wextra -Werror)

include(CTest)
if(BUILD_TESTING)
 find_package(Python3 QUIET COMPONENTS Interpreter)
 add_executable(client_digest_test tests/client_test.c)
 target_link_libraries(client_digest_test PRIVATE forge_storage)
 target_compile_options(client_digest_test PRIVATE -Wall -Wextra -Werror)
 if(Python3_Interpreter_FOUND)
  add_test(NAME client_digest_alias COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_client.py $<TARGET_FILE:client_digest_test>)
 endif()
endif()
